const writeContract = () => ({ ...contract, kind: 'implementation', mode: 'workspace-write', executionPlan: { allowedPaths: ['sum.js'], verification: { command: process.execPath, args: ['--input-type=module', '-e', "import assert from 'node:assert/strict'; import {sum} from './sum.js'; assert.equal(sum(2,3),5); console.log('verified sum');"], timeoutMs: 3000 } } });

test('NEW stale member progress, stale finish and stale review cannot contaminate a running retry', async t => {
  const calls = [], gates = [];
  const f = fixture(t, () => ({ run: req => { calls.push(req); const gate = deferred(); gates.push(gate); return gate.promise; } }));
  const task = f.runtime.create(contract), first = f.runtime.start(task.id); await tick();
  calls[0].onProgress({ memberLaunch: 'exited', actualModel: 'test-model', threadId: 'first-thread' });
  gates[0].resolve({ result: report(), actualModel: 'test-model' }); await f.runtime.wait(task.id);
  approve(f.runtime, task.id, first.attemptId, 'reject');
  const second = f.runtime.start(task.id, { retry: true, reason: 'Independent replay isolation probe' }); await tick();
  const before = f.runtime.status(task.id);
  calls[0].onProgress({ childPid: process.pid, memberLaunch: 'active', machineReview: { status: 'passed', actualModel: 'poison' }, threadId: 'STALE_POISON', usage: { poison: true } });
  assert.equal(f.runtime.finish(task.id, first.attemptId, 'succeeded', { result: { poison: true } }), false);
  assert.throws(() => approve(f.runtime, task.id, first.attemptId), /Stale/);
  const after = f.runtime.status(task.id);
  assert.deepEqual(after.attempts, before.attempts); assert.equal(after.status, 'running'); assert.equal(after.result, undefined);
  const events = f.runtime.store.db.prepare('SELECT kind FROM events WHERE task_id=? ORDER BY seq DESC LIMIT 2').all(task.id).map(r => r.kind);
  assert.deepEqual(events, ['finish-dropped', 'progress-dropped']);
  calls[1].onProgress({ memberLaunch: 'exited' }); gates[1].resolve({ result: report(), actualModel: 'test-model' });
  assert.equal((await f.runtime.wait(task.id)).status, 'awaiting_review');
  note('stale-callback', { first: first.attemptId, current: second.attemptId, events, poisoned: false });
});

test('NEW task+event transaction rolls back the task when durable event insertion fails', t => {
  const root = mkdtempSync(join(auditRoot, 'event-rollback-')), path = join(root, 'state.sqlite');
  const store = new TaskStore(path); t.after(() => store.close());
  store.save({ id: 'probe', status: 'pending', attempts: [] }, 'seed');
  const before = store.get('probe'), count = store.db.prepare('SELECT count(*) AS n FROM events').get().n;
  store.db.exec("CREATE TRIGGER fail_audit_event BEFORE INSERT ON events WHEN NEW.kind='probe-fail' BEGIN SELECT RAISE(ABORT,'controlled durable event failure'); END;");
  assert.throws(() => store.save({ ...before, status: 'succeeded' }, 'probe-fail'), /controlled durable event failure/);
  assert.deepEqual(store.get('probe'), before); assert.equal(store.db.prepare('SELECT count(*) AS n FROM events').get().n, count);
  note('event-rollback', { db: path, status: store.get('probe').status, eventCount: count });
});

test('NEW restart preserves awaiting_integration; identical-byte symlink logs reject integration; restored log allows explicit integration only', async t => {
  const f = fixture(t, writerBackend);
  writeFileSync(join(f.repo, 'user.txt'), 'staged user content\n'); f.git(['add', 'user.txt']);
  const head = f.git(['rev-parse', 'HEAD']), index = readFileSync(join(f.repo, '.git', 'index'));
  const spec = writeContract(); spec.executionPlan.dirtyPolicy = 'snapshot';
  const task = f.runtime.create(spec), dispatch = f.runtime.start(task.id);
  const ready = await f.runtime.wait(task.id, 15000); assert.equal(ready.status, 'awaiting_review', ready.error);
  approve(f.runtime, task.id, dispatch.attemptId);
  const dependent = f.runtime.create({ ...contract, dependencies: [task.id] }); assert.throws(() => f.runtime.start(dependent.id), /Dependencies/);
  await f.runtime.close(); f.open(); assert.equal(f.runtime.status(task.id).status, 'awaiting_integration');
  assert.match(readFileSync(join(f.repo, 'sum.js'), 'utf8'), /a-b/);
  const attempt = f.runtime.status(task.id).attempts[0], workspace = f.runtime.workspaceStatus(attempt.workspaceId);
  const check = workspace.verifications.find(v => v.id === attempt.verificationId), savedLog = check.stdoutPath + '.saved';
  renameSync(check.stdoutPath, savedLog); symlinkSync(savedLog, check.stdoutPath);
  assert.throws(() => f.runtime.integrate(task.id, dispatch.attemptId), /log evidence is missing or changed/);
  assert.equal(f.runtime.status(task.id).status, 'awaiting_integration'); assert.match(readFileSync(join(f.repo, 'sum.js'), 'utf8'), /a-b/);
  unlinkSync(check.stdoutPath); renameSync(savedLog, check.stdoutPath);
  f.runtime.integrate(task.id, dispatch.attemptId); const integrated = await f.runtime.wait(task.id, 15000);
  assert.equal(integrated.status, 'succeeded', integrated.error);
  assert.match(readFileSync(join(f.repo, 'sum.js'), 'utf8'), /a\+b/); assert.equal(f.git(['rev-parse', 'HEAD']), head);
  assert.deepEqual(readFileSync(join(f.repo, '.git', 'index')), index);
  f.runtime.start(dependent.id); assert.equal((await f.runtime.wait(dependent.id)).status, 'awaiting_review');
  note('persisted-integration', { db: f.path, symlinkRejected: true, status: integrated.status, headPreserved: true, indexBytesPreserved: true, dependentUnlockedAfterSourceVerification: true });
});
